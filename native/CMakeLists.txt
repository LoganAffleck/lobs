cmake_minimum_required(VERSION 3.21)
project(LOBS VERSION 0.6.5 LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJC OBJCXX)
endif()
# The OBS engine each release ships with and is tested against (build.bat installs it).
set(LOBS_ENGINE_VERSION 32.2.2)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Svg Network Widgets) # Widgets: plugin panels
# KDDockWidgets (QtQuick frontend), built into deps/kddw by deps/build-kddw.bat
find_package(KDDockWidgets-qt6 REQUIRED PATHS ${CMAKE_SOURCE_DIR}/deps/kddw NO_DEFAULT_PATH)
qt_standard_project_setup(REQUIRES 6.5)

# libobs headers (obs-studio 32.2.2). Windows links an import library generated from the bundled
# obs.dll; macOS links libobs.framework from the engine build.sh staged out of OBS's DMG.
set(LIBOBS_INCLUDE ${CMAKE_SOURCE_DIR}/../obs-plugin/deps/obs-studio/libobs)
set(LIBOBS_IMPLIB ${CMAKE_SOURCE_DIR}/deps/obs.lib)
set(LOBS_MAC_ENGINE ${CMAKE_SOURCE_DIR}/deps/engine-mac CACHE PATH "Engine staged from OBS's macOS DMG (Frameworks, PlugIns)")

qt_add_executable(LOBS WIN32 MACOSX_BUNDLE
    src/main.cpp
    src/platform.h
    src/permissions.cpp src/permissions.h
    src/app.cpp
    src/app.h
    src/engine.cpp
    src/engine_output.cpp
    src/engine_extras.cpp
    src/hotkeys.cpp
    src/shellmeta.h
    src/hotkeys.h
    src/settings.cpp
    src/settings.h
    src/icons.h
    src/engine.h
    src/engine_manager.h
    src/engine_version.cpp
    src/lobs_instance.c
    src/lobs_frame.c
    src/shapes.cpp
    src/eyedropper.cpp
    src/browser_view.cpp
    src/source_preview.cpp
    src/projector.cpp
    src/rows_model.cpp
    src/window_chrome.h
    src/captions.cpp src/captions.h
    src/twitch_account.cpp src/twitch_account.h
    src/enhanced_broadcast.cpp src/enhanced_broadcast.h
    src/obs_switch.cpp src/obs_switch.h
    src/websocket_server.cpp src/websocket_server.h
    src/frontend.cpp src/frontend.h
    src/plugin_panel.cpp src/plugin_panel.h
    src/plugins.cpp src/plugins.h
    src/crash_report.cpp src/crash_report.h
    src/app_updater.cpp src/app_updater.h
    src/dock_tabs.h
    src/rows_model.h
    src/projector.h
    src/browser_view.h
    src/source_preview.h
    src/eyedropper.h
    src/shapes.h
    src/model/doc.cpp
    src/model/doc.h
    src/model/store.cpp
    src/model/obs_import.cpp
    src/model/store.h
    src/model/video_fx.h
)
if(WIN32)
    target_sources(LOBS PRIVATE resources/lobs.rc src/platform_win.cpp src/window_chrome.cpp src/shellmeta.cpp
        src/crash_report_win.cpp src/engine_manager.cpp)
elseif(APPLE)
    target_sources(LOBS PRIVATE src/platform_mac.mm src/window_chrome_mac.mm src/shellmeta_mac.mm
        src/crash_report_mac.cpp src/engine_manager_mac.cpp)
endif()
qt_add_qml_module(LOBS URI LOBS VERSION 1.0 QML_FILES
    qml/Main.qml
    # Fluid Functionalism components (QML port)
    qml/FButton.qml qml/FInput.qml qml/FNumField.qml qml/FFillSlider.qml qml/FSwitch.qml qml/FCheckbox.qml
    qml/FTabs.qml qml/FMenu.qml qml/FMenuItem.qml qml/FSelect.qml qml/FDialog.qml qml/FluidList.qml
    qml/FSection.qml qml/FBadge.qml qml/FPanelHeader.qml qml/FCopyToMenu.qml qml/MediaControls.qml qml/AudioSettings.qml qml/AudioMath.js qml/InlineRename.qml qml/StatsView.qml qml/CaptionsPanel.qml
    qml/SettingsPage.qml qml/MixerView.qml qml/FIcon.qml qml/FHelp.qml qml/FScrollView.qml qml/CanvasArea.qml qml/StatusBar.qml qml/OutputControls.qml qml/AssetMouse.qml qml/MeterFader.qml qml/ToolTipStyle.qml qml/SourceFields.qml qml/BrowserOptions.qml qml/SourceSection.qml qml/FFontPicker.qml qml/FColorField.qml qml/FSearchField.qml qml/GraphicTools.qml
        qml/TransitionsDialog.qml
        qml/ObsProperties.qml
        qml/NewProjectDialog.qml
        qml/SetupWizard.qml
        qml/TwitchPanel.qml
        qml/PermissionsBar.qml
    qml/dock/DockTabBar.qml qml/dock/DockTitleBar.qml qml/dock/DockGroup.qml qml/dock/DockFloatingWindow.qml qml/dock/DockSeparator.qml qml/dock/DockWidgetBg.qml qml/dock/PanelDragController.qml
    RESOURCES resources/icon/lobs-32.png resources/icon/lobs-64.png)

target_include_directories(LOBS PRIVATE ${LIBOBS_INCLUDE})
target_include_directories(LOBS PRIVATE ${CMAKE_SOURCE_DIR}/deps/obs-frontend-api) # OBS 32.2.2's frontend API headers
# obsconfig.h is generated by an OBS build; LOBS only needs its release flags.
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file(${LIBOBS_INCLUDE}/obsconfig.h.in ${CMAKE_BINARY_DIR}/obsconfig/obsconfig.h)
target_include_directories(LOBS PRIVATE ${CMAKE_BINARY_DIR}/obsconfig)
if(APPLE)
    # libobs's SSE code goes through SIMDe on Apple Silicon (header-only, deps/simde).
    target_include_directories(LOBS PRIVATE ${CMAKE_SOURCE_DIR}/deps/simde)
endif()
# whisper.cpp (live captions), built by deps/build-whisper.bat (.sh on macOS)
target_include_directories(LOBS PRIVATE ${CMAKE_SOURCE_DIR}/deps/whisper/include)
target_compile_definitions(LOBS PRIVATE LOBS_VERSION="${PROJECT_VERSION}" LOBS_ENGINE_VERSION="${LOBS_ENGINE_VERSION}")
target_link_libraries(LOBS PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Svg Qt6::Network Qt6::Widgets KDAB::kddockwidgets)
if(WIN32)
    target_compile_definitions(LOBS PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(LOBS PRIVATE ${CMAKE_SOURCE_DIR}/deps/whisper/lib/whisper.lib ${LIBOBS_IMPLIB}
        crypt32 version delayimp shell32 ole32 propsys dwmapi dbghelp wintrust d3d11 dxgi)
    # libobs is loaded from the engine folder chosen at startup (<app>/engine/<version>), so it
    # must not be resolved before main() runs.
    target_link_options(LOBS PRIVATE "/DELAYLOAD:obs.dll")
    # App folder (what the installer ships): dist/LOBS/LOBS.exe, engines in dist/LOBS/engine/<version>
    set_target_properties(LOBS PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/dist/LOBS)
elseif(APPLE)
    # libobs's graphics headers hold an Objective-C object (gs_window.view) on macOS, so C++
    # that includes obs.h builds as Objective-C++, as OBS's own frontend does.
    get_target_property(_lobs_sources LOBS SOURCES)
    list(FILTER _lobs_sources INCLUDE REGEX "\\.cpp$")
    set_source_files_properties(${_lobs_sources} PROPERTIES LANGUAGE OBJCXX)
    target_compile_options(LOBS PRIVATE -F${LOBS_MAC_ENGINE}/Frameworks $<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fobjc-arc>)
    target_link_options(LOBS PRIVATE -F${LOBS_MAC_ENGINE}/Frameworks)
    target_link_libraries(LOBS PRIVATE "-framework libobs" ${CMAKE_SOURCE_DIR}/deps/whisper/lib/libwhisper.dylib
        "-framework AppKit" "-framework AVFoundation" "-framework CoreMedia" "-framework QuickLookThumbnailing"
        "-framework Security" "-framework Carbon")
    # LOBS.app: libobs, its dylibs and the CEF helpers in Contents/Frameworks, OBS's modules in
    # Contents/PlugIns, where libobs looks for them (build.sh copies them in).
    set_target_properties(LOBS PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/dist/mac
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_SOURCE_DIR}/resources/mac/Info.plist.in
        MACOSX_BUNDLE_BUNDLE_NAME LOBS
        MACOSX_BUNDLE_GUI_IDENTIFIER com.loganaffleck.lobs
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
        INSTALL_RPATH "@executable_path/../Frameworks"
        BUILD_WITH_INSTALL_RPATH ON)
endif()

# QA build (qa/build-qa.bat): adds the --scenariotest runner and goes to dist/LOBS-qa instead,
# so the shipped app never contains it.
option(LOBS_TESTS "Build the QA scenario runner" OFF)
if(LOBS_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    target_sources(LOBS PRIVATE src/test_runner.cpp src/test_runner.h)
    target_compile_definitions(LOBS PRIVATE LOBS_TESTS QT_QML_DEBUG) # QML profiler for perf work (QA build only)
    target_link_libraries(LOBS PRIVATE Qt6::Test)
    if(WIN32)
        target_link_libraries(LOBS PRIVATE psapi)
        set_target_properties(LOBS PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/dist/LOBS-qa)
    else()
        set_target_properties(LOBS PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/dist/mac-qa)
    endif()
endif()
